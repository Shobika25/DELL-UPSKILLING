case class Candidate(
    name: String,
    score: Option[Int]
)

object CandidateAssessment {


  def passedCandidates(
      candidates: List[Candidate]
  ): List[String] = {

    candidates.filter {
      case Candidate(name, Some(score)) if score >= 60 => true
      case _ => false
    }.map(_.name)
  }


  def failedCandidates(
      candidates: List[Candidate]
  ): List[String] = {

    candidates.filter {
      case Candidate(name, Some(score)) if score < 60 => true
      case _ => false
    }.map(_.name)
  }


  def pendingCandidates(
      candidates: List[Candidate]
  ): List[String] = {

    candidates.filter {
      case Candidate(name, None) => true
      case _ => false
    }.map(_.name)
  }


  def averageScore(
      candidates: List[Candidate]
  ): Double = {

    val scores = candidates.flatMap {
      case Candidate(name, Some(score)) => Some(score)
      case Candidate(name, None) => None
    }

    if (scores.isEmpty) 0.0
    else scores.map(_.toDouble).sum / scores.size
  }

  def main(args: Array[String]): Unit = {

    val candidates = List(
      Candidate("John", Some(85)),
      Candidate("David", Some(45)),
      Candidate("Smith", None),
      Candidate("Peter", Some(75)),
      Candidate("Alex", None)
    )

    println("Passed candidates: " +
      passedCandidates(candidates).mkString(", "))

    println("Failed candidates: " +
      failedCandidates(candidates).mkString(", "))

    println("Pending candidates: " +
      pendingCandidates(candidates).mkString(", "))

    println(f"Average score: ${averageScore(candidates)}%.2f")
  }
}

case class Employee(
    id: Int,
    name: String,
    department: String,
    salary: Option[Double]
)

object EmployeeSalaryValidation {

  def validateSalary(
      employee: Employee
  ): Either[String, Double] = {

    employee.salary match {

      case Some(salary) if salary > 0 =>
        Right(salary)

      case Some(salary) =>
        Left(s"${employee.name} has invalid salary: $salary")

      case None =>
        Left(s"${employee.name} has missing salary")
    }
  }

  def main(args: Array[String]): Unit = {

    val employees = List(
      Employee(101, "John", "IT", Some(50000)),
      Employee(102, "David", "HR", Some(30000)),
      Employee(103, "Smith", "IT", None),
      Employee(104, "Peter", "Finance", Some(-1000)),
      Employee(105, "Alex", "IT", Some(70000))
    )


    val validationResults = employees.map { employee =>
      (employee, validateSalary(employee))
    }


    val validEmployees = validationResults.filter {
      case (employee, Right(salary)) => true
      case (employee, Left(error)) => false
    }


    val invalidEmployees = validationResults.filter {
      case (employee, Left(error)) => true
      case (employee, Right(salary)) => false
    }


    val validITEmployees = validEmployees.filter {
      case (employee, result) =>
        employee.department == "IT"
    }


    val itSalaries = validITEmployees.map {
      case (employee, Right(salary)) => salary
      case _ => 0.0
    }

    val averageITSalary =
      if (itSalaries.isEmpty) 0.0
      else itSalaries.sum / itSalaries.size


    println("Valid employees: " +
      validEmployees.map(_._1.name).mkString(", "))

    println("Invalid employees: " +
      invalidEmployees.map(_._1.name).mkString(", "))

    println("Valid IT employees: " +
      validITEmployees.map(_._1.name).mkString(", "))

    println(f"Average IT salary: $averageITSalary%.0f")
  }
}

def validateMarks(marks: Option[Int]): Either[String, Int] = {
  marks match {
    case None => Left("Marks are missing")
    case Some(value) if value < 0 || value > 100 =>
      Left("Marks must be between 0 and 100")
    case Some(value) => Right(value)
  }
}


def calculateTotal(student: Student): Int = {
  student.subjects.flatMap(_.marks).sum
}


val incompleteStudents = students.filter {
  case Student(_, _, subjects) =>
    subjects.exists {
      case Subject(_, None) => true
      case Subject(_, Some(_)) => false
    }
}


val completeStudents = students.filter {
  student =>
    student.subjects.forall {
      subject => validateMarks(subject.marks).isRight
    }
}


def percentage(student: Student): Double = {
  calculateTotal(student).toDouble /
    (student.subjects.size * 100) * 100
}


val passedStudents = completeStudents.filter {
  student => percentage(student) >= 50
}


val distinctionStudents = completeStudents.filter {
  student => percentage(student) >= 75
}

def validateCustomer(
  customer: Customer
): Either[String, Customer] = {

  customer match {

    case Customer(None, _, _, _) =>
      Left("Customer ID is missing")

    case Customer(Some(id), _, _, _) if id <= 0 =>
      Left("Customer ID must be positive")

    case Customer(_, None, _, _) =>
      Left("Customer name is missing")

    case Customer(_, Some(name), _, _) if name.trim.isEmpty =>
      Left("Customer name cannot be empty")

    case Customer(_, _, None, _) =>
      Left("Customer age is missing")

    case Customer(_, _, Some(age), _) if age < 18 || age > 100 =>
      Left("Age must be between 18 and 100")

    case Customer(_, _, _, None) =>
      Left("Email is missing")

    case Customer(_, _, _, Some(email)) if !email.contains("@") =>
      Left("Invalid email address")

    case validCustomer =>
      Right(validCustomer)
  }
}
val validationResults = customers.map(validateCustomer)


val validCustomers = validationResults.flatMap {
  case Right(customer) => Some(customer)
  case Left(error) => None
}


val rejectedCustomers = customers.zip(validationResults).collect {
  case (customer, Left(error)) => (customer, error)
}


println("Total records received: " + customers.size)

println("Valid customer count: " + validCustomers.size)
